/* =========================================================

   TEXT FORMATTING

   ========================================================= */


function normalizeText(value) {


  return String(


    value == null

      ?

      ""

      :

      value


  )


    .replace(

      /[\u2018\u2019]/g,

      "'"

    )


    .replace(

      /[\u201C\u201D]/g,

      '"'

    )


    .toUpperCase()


    .slice(

      0,

      COLUMNS

    );


}




function centerText(value) {


  const text =

    normalizeText(

      value

    );



  const remaining =

    COLUMNS

    -

    text.length;



  const left =

    Math.floor(

      remaining / 2

    );



  const right =

    remaining - left;



  return (


    " ".repeat(left)


    +


    text


    +


    " ".repeat(right)


  );


}




function formatLines(lines) {


  const output =

    [];



  for (

    let row = 0;

    row < ROWS;

    row++

  ) {


    const currentLine =


      (

        lines

        &&

        lines[row] != null

      )


      ?


      lines[row]


      :


      "";



    const centered =

      centerText(

        currentLine

      );



    const characters =

      centered.split(

        ""

      );



    for (

      let i = 0;

      i < characters.length;

      i++

    ) {


      output.push(

        characters[i]

      );


    }


  }



  if (

    output.length

    !==

    TILE_COUNT

  ) {


    throw new Error(


      "Expected "

      +

      TILE_COUNT

      +

      " tiles but got "

      +

      output.length


    );


  }



  return output;


}

function clearColors() {

function applyColors(colors) {


  clearColors();



  if (

    !Array.isArray(

      colors

    )

  ) {


    return;


  }



  for (

    let i = 0;

    i < colors.length;

    i++

  ) {


    const entry =

      colors[i];



    const row =

      Number(

        entry

        &&

        entry.row

      );



    const col =

      Number(

        entry

        &&

        entry.col

      );



    const color =

      String(


        (

          entry

          &&

          entry.color

        )


        ||


        ""


      )

        .toLowerCase();



    if (


      !Number.isInteger(row)


      ||


      !Number.isInteger(col)


    ) {


      continue;


    }



    if (


      row < 0


      ||


      row >= ROWS


      ||


      col < 0


      ||


      col >= COLUMNS


    ) {


      continue;


    }



    if (

      !VALID_COLORS.has(

        color

      )

    ) {


      continue;


    }



    tiles[

      row * COLUMNS + col

    ]

      .tile

      .classList

      .add(

        "color-" + color

      );


  }


}

function animateTile(

  if (

    !item

    item.character.textContent

    ===

    finalCharacter

  ) {


    return;


  }



  window.setTimeout(


    function () {



      if (

        generation

        !==

        animationGeneration

      ) {


        return;


      }



      const totalFlips =


        MIN_FLIPS


        +


        Math.floor(


          Math.random()


          *


          (

            EXTRA_FLIPS + 1

          )


        );



      let flipCount =

        0;



      const interval =


        window.setInterval(


          function () {



            if (

              generation

              !==

              animationGeneration

            ) {


              window.clearInterval(

                interval

              );


              return;


            }



            item

              .tile

              .classList

              .remove(

                "flipping"

              );



            void item

              .tile

              .offsetWidth;



            item

              .tile

              .classList

              .add(

                "flipping"

              );



            flipCount++;



            if (

              flipCount

              >=

              totalFlips

            ) {


              window.clearInterval(

                interval

              );



              item

                .character

                .textContent =

                  finalCharacter;



              item

                .tile

                .classList

                .remove(

                  "flipping"

                );



              return;


            }



            item

              .character

              .textContent =

                randomFlipCharacter();


          },


          FLIP_SPEED_MS


        );


    },


    delay


  );


}




/* =========================================================

  if (

    !scene

  animationGeneration++;



  const generation =

  const formatted =

  applyColors(

  for (

    let index = 0;

    index < formatted.length;

    index++

  ) {


    const row =

      Math.floor(

        index / COLUMNS

      );



    const col =

      index % COLUMNS;



    const delay =


      col * 11


      +


      row * 33


      +


      Math.random() * 70;



    animateTile(

/* =========================================================

   RANDOM MESSAGE WITHOUT REPEATING

   ========================================================= */


function randomIndexAvoidRepeat(


  length,


  lastIndex


) {


  if (

    length <= 1

  ) {


    return 0;


  }



  let index =

    Math.floor(

      Math.random() * length

    );



  if (

    index === lastIndex

  ) {


    index =


      (

        index


        +


        1


        +


        Math.floor(

          Math.random()

          *

          (length - 1)

        )


      )


      %


      length;


  }



  return index;


}

        afternoonMessages,

        "evening",


      messages:

        eveningMessages,

        "orange"


    };

  return {

      nightMessages,

      "purple"


  };


}

/* =========================================================

   ROTATION LOGIC

   ========================================================= */


function getNextScene() {


  const special =

    getSpecialDateScene();



  if (

    special

    &&

    rotationStep % 4 === 0

  ) {


    rotationStep++;


    return special;


  }



  /*

     Normal repeating pattern:


     time-based message

     date/time

     quote

     time-based message

     quote

  */


  const pattern = [


    "time",


    "date",


    "quote",


    "time",


    "quote"


  ];



  const type =


    pattern[

      rotationStep

      %

      pattern.length

    ];



  rotationStep++;



  if (

    type === "date"

  ) {


    return buildDateTimeScene();


  }



  if (

    type === "quote"

  ) {


    return buildQuoteScene();


  }



  return buildTimeMessageScene();


}

/* =========================================================

   FULLSCREEN

   ========================================================= */


function enterFullscreen() {


  const element =

    document.documentElement;



  if (

    document.fullscreenElement

  ) {


    return;


  }



  if (

    typeof element.requestFullscreen

    ===

    "function"

  ) {


    try {


      const result =

        element.requestFullscreen();



      if (


        result


        &&


        typeof result.then

        ===

        "function"


      ) {


        result


          .then(

            hideFullscreenHint

          )


          .catch(

            function () {}

          );


      }


      else {


        hideFullscreenHint();


      }


    }


    catch (error) {}


    return;


  }



  if (

    typeof element.webkitRequestFullscreen

    ===

    "function"

  ) {


    try {


      element

        .webkitRequestFullscreen();



      hideFullscreenHint();


    }


    catch (error) {}


  }


}

/* =========================================================

   WAKE LOCK

   ========================================================= */


function requestWakeLock() {


  if (


    !navigator


    ||


    !navigator.wakeLock


    ||


    typeof navigator.wakeLock.request

    !==

    "function"


  ) {


    return;


  }



  if (


    wakeLock


    &&


    wakeLock.released === false


  ) {


    return;


  }



  navigator

    .wakeLock

    .request(

      "screen"

    )


    .then(


      function (lock) {


        wakeLock =

          lock;



        if (


          wakeLock


          &&


          typeof wakeLock.addEventListener

          ===

          "function"


        ) {


          wakeLock

            .addEventListener(


              "release",


              function () {


                wakeLock =

                  null;


              }


            );


        }


      }


    )


    .catch(

      function () {}

    );


}
